cmake_minimum_required(VERSION 3.20)
project(MeetOverlay VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(GNUInstallDirs)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Quick Qml Network Test)
find_package(LayerShellQt REQUIRED)
qt_standard_project_setup()
add_library(native_framing src/native_framing.cpp)
target_include_directories(native_framing PUBLIC src)
target_link_libraries(native_framing PUBLIC Qt6::Core)
add_executable(meetoverlay-host src/host.cpp)
target_link_libraries(meetoverlay-host PRIVATE native_framing Qt6::Network)
add_executable(meetoverlay src/main.cpp src/controller.cpp src/controller.h)
target_link_libraries(meetoverlay PRIVATE Qt6::Gui Qt6::Quick Qt6::Qml Qt6::Network LayerShellQt::Interface)
qt_add_resources(meetoverlay qml_resources PREFIX "/" FILES qml/Overlay.qml qml/Settings.qml qml/Indicator.qml)
enable_testing()
add_executable(native_framing_test tests/native/native_framing_test.cpp)
target_link_libraries(native_framing_test PRIVATE native_framing Qt6::Test)
add_test(NAME native_framing COMMAND native_framing_test)
install(TARGETS meetoverlay RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(TARGETS meetoverlay-host RUNTIME DESTINATION ${CMAKE_INSTALL_LIBDIR}/meetoverlay)
install(DIRECTORY extension/ DESTINATION ${CMAKE_INSTALL_DATADIR}/meetoverlay/extension)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/meetoverlay)
add_executable(controller_test tests/integration/controller_test.cpp src/controller.cpp src/controller.h)
target_include_directories(controller_test PRIVATE src)
target_link_libraries(controller_test PRIVATE Qt6::Gui Qt6::Quick Qt6::Network Qt6::Test LayerShellQt::Interface)
add_test(NAME controller COMMAND controller_test)
set_tests_properties(controller PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME native_host COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/host_test.py $<TARGET_FILE:meetoverlay-host>)
configure_file(data/org.meetoverlay.host.json.in org.meetoverlay.host.json @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/org.meetoverlay.host.json DESTINATION /etc/opt/chrome/native-messaging-hosts)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/org.meetoverlay.host.json DESTINATION /etc/chromium/native-messaging-hosts)
install(FILES data/meetoverlay.service DESTINATION ${CMAKE_INSTALL_LIBDIR}/systemd/user)
install(FILES data/meetoverlay.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES data/meetoverlay-autostart.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/meetoverlay)
