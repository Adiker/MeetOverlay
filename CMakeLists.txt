cmake_minimum_required(VERSION 3.20)
project(MeetOverlayResearch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Test)
qt_standard_project_setup()
add_library(native_framing src/native_framing.cpp)
target_include_directories(native_framing PUBLIC src)
target_link_libraries(native_framing PUBLIC Qt6::Core)
enable_testing()
add_executable(native_framing_test tests/native/native_framing_test.cpp)
target_link_libraries(native_framing_test PRIVATE native_framing Qt6::Test)
add_test(NAME native_framing COMMAND native_framing_test)
